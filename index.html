#include #include // =============================== // WIFI // =============================== const char* WIFI_SSID = "YOUR_WIFI_NAME"; const char* WIFI_PASSWORD = "YOUR_WIFI_PASSWORD"; // =============================== // MQTT // =============================== const char* MQTT_SERVER = "broker.hivemq.com"; const int MQTT_PORT = 1883; const char* TOPIC_FORWARD = "toycar/8f3a91c7e2b64d10/forward"; const char* TOPIC_BACKWARD = "toycar/8f3a91c7e2b64d10/backward"; const char* TOPIC_LEFT = "toycar/8f3a91c7e2b64d10/left"; const char* TOPIC_RIGHT = "toycar/8f3a91c7e2b64d10/right"; const char* TOPIC_BRAKE = "toycar/8f3a91c7e2b64d10/brake"; const char* TOPIC_SPEED = "toycar/8f3a91c7e2b64d10/speed"; // =============================== // ESP32 PINS // =============================== const int FORWARD_PIN = 23; const int BACKWARD_PIN = 22; const int LEFT_PIN = 21; const int RIGHT_PIN = 19; const int BRAKE_PIN = 18; const int SPEED_PIN = 27; // =============================== // MQTT OBJECTS // =============================== WiFiClient wifiClient; PubSubClient mqtt(wifiClient); // =============================== // FAILSAFE TIMER // =============================== unsigned long lastMotionMessage = 0; const unsigned long FAILSAFE_TIME = 1000; // =============================== // SPEED PULSE // =============================== bool speedPulseActive = false; unsigned long speedPulseStart = 0; const unsigned long SPEED_PULSE_TIME = 200; // =============================== // STOP MOTION // =============================== void stopMotion() { digitalWrite(FORWARD_PIN, LOW); digitalWrite(BACKWARD_PIN, LOW); digitalWrite(LEFT_PIN, LOW); digitalWrite(RIGHT_PIN, LOW); } // =============================== // STOP EVERYTHING // =============================== void allStop() { stopMotion(); digitalWrite(BRAKE_PIN, LOW); digitalWrite(SPEED_PIN, LOW); speedPulseActive = false; } // =============================== // SPEED BUTTON PULSE // =============================== void startSpeedPulse() { if (!speedPulseActive) { digitalWrite(SPEED_PIN, HIGH); speedPulseStart = millis(); speedPulseActive = true; Serial.println("SPEED BUTTON PRESS"); } } // =============================== // MQTT CALLBACK // =============================== void callback( char* topic, byte* payload, unsigned int length ) { String message = ""; for ( unsigned int i = 0; i < length; i++ ) { message += (char)payload[i]; } String topicString = String(topic); Serial.print("Topic: "); Serial.println(topicString); Serial.print("Message: "); Serial.println(message); // =============================== // FORWARD // =============================== if ( topicString == TOPIC_FORWARD ) { lastMotionMessage = millis(); if ( message == "1" ) { digitalWrite( BACKWARD_PIN, LOW ); digitalWrite( FORWARD_PIN, HIGH ); Serial.println( "FORWARD ON" ); } else { digitalWrite( FORWARD_PIN, LOW ); } } // =============================== // BACKWARD // =============================== else if ( topicString == TOPIC_BACKWARD ) { lastMotionMessage = millis(); if ( message == "1" ) { digitalWrite( FORWARD_PIN, LOW ); digitalWrite( BACKWARD_PIN, HIGH ); Serial.println( "BACKWARD ON" ); } else { digitalWrite( BACKWARD_PIN, LOW ); } } // =============================== // LEFT // =============================== else if ( topicString == TOPIC_LEFT ) { lastMotionMessage = millis(); if ( message == "1" ) { digitalWrite( RIGHT_PIN, LOW ); digitalWrite( LEFT_PIN, HIGH ); Serial.println( "LEFT ON" ); } else { digitalWrite( LEFT_PIN, LOW ); } } // =============================== // RIGHT // =============================== else if ( topicString == TOPIC_RIGHT ) { lastMotionMessage = millis(); if ( message == "1" ) { digitalWrite( LEFT_PIN, LOW ); digitalWrite( RIGHT_PIN, HIGH ); Serial.println( "RIGHT ON" ); } else { digitalWrite( RIGHT_PIN, LOW ); } } // =============================== // BRAKE // =============================== else if ( topicString == TOPIC_BRAKE ) { if ( message == "1" ) { // Brake overrides movement stopMotion(); digitalWrite( BRAKE_PIN, HIGH ); Serial.println( "BRAKE ON" ); } else { digitalWrite( BRAKE_PIN, LOW ); Serial.println( "BRAKE OFF" ); } } // =============================== // SPEED INCREASE // =============================== else if ( topicString == TOPIC_SPEED ) { if ( message == "1" ) { startSpeedPulse(); } } } // =============================== // CONNECT MQTT // =============================== void connectMQTT() { while ( !mqtt.connected() ) { Serial.println( "Connecting to MQTT..." ); String clientId = "ToyCarESP32-" + String( random(0xffff), HEX ); if ( mqtt.connect( clientId.c_str() ) ) { Serial.println( "MQTT connected!" ); mqtt.subscribe( TOPIC_FORWARD ); mqtt.subscribe( TOPIC_BACKWARD ); mqtt.subscribe( TOPIC_LEFT ); mqtt.subscribe( TOPIC_RIGHT ); mqtt.subscribe( TOPIC_BRAKE ); mqtt.subscribe( TOPIC_SPEED ); Serial.println( "Subscribed to all 6 controls" ); } else { Serial.print( "MQTT failed. Code: " ); Serial.println( mqtt.state() ); delay(2000); } } } // =============================== // SETUP // =============================== void setup() { Serial.begin(115200); pinMode( FORWARD_PIN, OUTPUT ); pinMode( BACKWARD_PIN, OUTPUT ); pinMode( LEFT_PIN, OUTPUT ); pinMode( RIGHT_PIN, OUTPUT ); pinMode( BRAKE_PIN, OUTPUT ); pinMode( SPEED_PIN, OUTPUT ); allStop(); WiFi.begin( WIFI_SSID, WIFI_PASSWORD ); Serial.print( "Connecting to WiFi" ); while ( WiFi.status() != WL_CONNECTED ) { delay(500); Serial.print("."); } Serial.println(); Serial.println( "WiFi connected!" ); Serial.print( "ESP32 IP: " ); Serial.println( WiFi.localIP() ); mqtt.setServer( MQTT_SERVER, MQTT_PORT ); mqtt.setCallback( callback ); lastMotionMessage = millis(); } // =============================== // LOOP // =============================== void loop() { if ( !mqtt.connected() ) { allStop(); connectMQTT(); } mqtt.loop(); // =============================== // SPEED PULSE END // =============================== if ( speedPulseActive && millis() - speedPulseStart >= SPEED_PULSE_TIME ) { digitalWrite( SPEED_PIN, LOW ); speedPulseActive = false; Serial.println( "SPEED BUTTON RELEASE" ); } // =============================== // MOTION FAILSAFE // =============================== if ( millis() - lastMotionMessage > FAILSAFE_TIME ) { stopMotion(); } }
